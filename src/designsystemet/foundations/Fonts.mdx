# Fonts

## Overview

Here you will find an overview of the different fonts that are available.

Use the theme token to access the sans-serif font by using: (`--ds-font-family`). The monospace font is exposed as a Tailwind utility, `font-jetbrains-mono`, backed by the `--ads-font-jetbrains-mono` CSS variable.

export const fontTokens = [
    {
        name: "Inter",
        description: "Default sans-serif font used throughout the design system.",
        cssVar: "var(--ds-font-family)",
        sample: "ads:font-sans"
    },
    {
        name: "JetBrains Mono",
        description: "Monospace font used for code, technical and tabular content.",
        cssVar: "var(--ads-font-jetbrains-mono)",
        sample: "ads:font-jetbrains-mono"
    }
];

<div
  className="
    ads:flex
    ads:flex-col
    ads:gap-10
    ads:p-8
  "
>
  {fontTokens.map((token) => (
    <div
      key={token.name}
      className="
        ads:flex
        ads:flex-col
        ads:gap-4
        ads:p-6
        ads:rounded-[var(--ds-border-radius-md)]
        ads:bg-[var(--ds-color-surface-default)]
      "
    >
      <div className="ads:text-sm">
        {token.name}
      </div>
      <div className="ads:text-sm ads:text-[var(--ds-color-neutral-text-subtle)]">
        {token.description}
      </div>
      <div
        className="ads:text-3xl"
        style={{ fontFamily: token.cssVar }}
      >
        Aa Bb Cc 123
      </div>
    </div>

))}

</div>
